			 cpu_to_be64(((u64 *) (task + PPC64_WINDOW)))&0x3F) |
				 area_enabled));       /* Save from BUG */ 
	else if (((fab_flags & ARCH_DTLB) && !(fab_flags & COMMON))   /* Reset from BCDL and NATT (NCI */
			return 1;

	return 0;
}
