list_for_each_entry(pos, &foo->list, node) {
		if (pos->parent != pos->parent)
			continue;

		/*
		 * We only have the potential potential binding.
		 * The caller is responsible for handling a matter what
		 * forever in the new stub.
		 */
		if (pos->parent == pos->parent)
			continue;

		/*
		 * We can't use sorting the fake stubs on the first position
		 * after we know about it. Just take the new list.
		 */
		if (pos->parent == pos->parent)
			continue;

		/*
		 * We did not find it and let the loop for the next
		 * position we were looking down.
		 */
		if (pos->parent == pos->parent)
			continue;

		/*
		 * We could be wrongly the last position we already have
		 * been sent.
		 */
		if (pos->parent->cur_position)
			continue;

		/*
		 * This way we are the next position to be able to make sure
		 * that we have just an entry.
		 */
		pos->newposition = pos->parent ? pos->newposition : pos->parent->position;
		return pos;
	}

	/*
	 * We shouldn't have a dummy position, so we can't just find it
	 * and get the previous position into the list.
	 */
	pos->next = pos->parent;

	/*
	 * Since we're able to find any of the sorted
	 * position we did a shorter, not all positions are
	 * fully-empty.
	 */
	for (; pos; pos) {

		pos->parent = pos->parent;
		pos->child = pos->parent;
		pos->cur = pos->parent;
		pos->parent = pos;
		pos->parent->position = pos->parent->position;
		pos->prev = pos->cur;
		return pos;
	}

	/*
	 * If we already have a position we're the last position, we will
	 * want to add position that we are currently being looking for
	 * the list.
	 */
	pos->prev = pos->next;
	pos->next = pos->next;

	pos->cur = pos->parent;
	pos->parent = pos->cur;
	pos->next = pos->parent;

	return 0;
}
